/* Private, no-dependency Markdown publishing editor. Cloudflare Access protects every request. */
(function(){
 'use strict';
 var form=document.getElementById('editor-form');
 var listNode=document.getElementById('article-list');
 var filter=document.getElementById('filter-articles');
 var preview=document.getElementById('preview-content');
 var status=document.getElementById('editor-status');
 var message=document.getElementById('message');
 var count=document.getElementById('count');
 var draftButton=document.getElementById('save-draft');
 var publishButton=document.getElementById('publish');
 var note=document.getElementById('legacy-note');
 var revisionPanel=document.getElementById('revisions-panel');
 var revisionList=document.getElementById('revisions-list');
 var rows=[],active=null,typingTimer=null,manualSlug=false,previewVersion=0,busy=false;
 var fieldNames=['slug','title','summary','category','cover_url','body_md','seo_title','seo_description'];
 function field(name){return form.elements.namedItem(name);}
 function inform(t,problem){
  message.textContent=t;
  message.style.color=problem?'#f0b9a9':'#a6dcc0';
 }
 function safeText(s){return String(s===undefined||s===null?'':s);}
 function api(path,options){
  return fetch(path,Object.assign({credentials:'same-origin',headers:{'Accept':'application/json'}},options||{}))
   .then(function(r){return r.json().then(function(d){if(!r.ok)throw Error(d.error||'Request failed');return d;});});
 }
 function updateCount(){var text=field('body_md').value.trim();document.getElementById('word-count').textContent=(text?text.split(/\s+/).length:0)+' words';}
 function slugify(t){return safeText(t).toLowerCase().replace(/[^a-z0-9]+/g,'-').replace(/^-+|-+$/g,'').slice(0,95);}
 function setUrl(){
  var name=field('slug').value.trim()||'your-article-title';
  var label=active&&active.content_type==='legacy'?'Existing HTML: '+active.url:'blog.nawazsharif.in/'+name;
  document.getElementById('url-preview').textContent=label;
 }
 function header(){
  var isLegacy=!!(active&&active.content_type==='legacy');
  document.getElementById('editor-label').textContent=isLegacy?'EXISTING HTML ARTICLE':active?'EDITING ARTICLE':'NEW DYNAMIC ARTICLE';
  document.getElementById('editor-title').textContent=active?'Edit: '+active.title:'Write a new article';
  document.getElementById('editor-context').textContent=isLegacy?'Original HTML stays as fallback. Save draft does not change live content.':'Save draft without changing your published article.';
  status.textContent=active&&active.is_published?'Published':'Draft';
  status.className='status-tag'+(active&&active.is_published?'':' draft');
  note.hidden=!isLegacy;
  field('slug').disabled=!!active;
  draftButton.textContent='Save draft';
  publishButton.textContent=isLegacy?'Save & publish':'Publish article';
  revisionPanel.hidden=!active;
  setUrl();
 }
 function blank(){
  active=null;manualSlug=false;form.reset();
  field('slug').disabled=false;
  field('category').value='Technology';
  field('body_md').value='';
  field('content_override').checked=false;
  preview.replaceChildren();
  revisionList.replaceChildren();
  revisionPanel.open=false;
  inform('');
  header();updateCount();drawList();schedulePreview();
 }
 function setArticle(row){
  active=row;manualSlug=true;
  fieldNames.forEach(function(name){field(name).value=safeText(row[name]);});
  field('content_override').checked=!!row.content_override;
  header();updateCount();drawList();schedulePreview();
  inform('Editing '+row.slug+'. Changes are not saved until you click a button.');
  loadHistory();
 }
 function drawList(){
  listNode.replaceChildren();
  var query=filter.value.trim().toLowerCase(),shown=0;
  rows.forEach(function(row){
   if(query&&!(row.title+' '+row.category+' '+row.slug).toLowerCase().includes(query))return;
   shown++;
   var button=document.createElement('button');
   button.type='button';button.className='item'+(active&&active.slug===row.slug?' active':'');
   var title=document.createElement('strong');title.textContent=row.title;
   var meta=document.createElement('small');
   var tag=document.createElement('span');tag.className='status'+(!row.is_published?' draft':'');
   tag.textContent=row.is_published?'Published':'Draft';
   meta.appendChild(tag);
   meta.appendChild(document.createTextNode(' · '+(row.content_type==='legacy'?'Original HTML':'Dynamic')+' · '+row.category));
   button.appendChild(title);button.appendChild(meta);
   button.addEventListener('click',function(){setArticle(row);document.getElementById('title').focus();});
   listNode.appendChild(button);
  });
  count.textContent=shown+' of '+rows.length+' articles';
  if(!shown){var none=document.createElement('p');none.className='subtle';none.textContent='No matching articles.';listNode.appendChild(none);}
 }
 function refresh(focusSlug) {
  return api('/admin/api/articles').then(function(data){
   rows=data.articles||[];
   if(focusSlug){
    var found=rows.find(function(r){return r.slug===focusSlug;});
    if(found)setArticle(found);else blank();
   }else drawList();
   return rows;
  });
 }
 function schedulePreview(){
  updateCount();
  if(typingTimer)clearTimeout(typingTimer);
  if(!field('body_md').value.trim()){preview.textContent='Write Markdown to preview your article.';return;}
  var version=++previewVersion;
  typingTimer=setTimeout(function(){
   api('/admin/api/preview',{method:'POST',headers:{'Content-Type':'application/json','Accept':'application/json'},body:JSON.stringify({body_md:field('body_md').value})})
    .then(function(data){if(version!==previewVersion)return;preview.innerHTML=data.html;})
    .catch(function(e){if(version!==previewVersion)return;preview.textContent='Preview unavailable: '+e.message;});
  },380);
 }
 function readForm(publish){
  var entry={};
  fieldNames.forEach(function(name){entry[name]=field(name).value.trim();});
  entry.content_override=!!field('content_override').checked;
  entry.intent=publish?'publish':'draft';
  return entry;
 }
 function onSave(event){
  event.preventDefault();
  if(busy)return;
  var submit=event.submitter || document.activeElement;
  var wantPublish=submit&&submit.id==='publish';
  var isLegacy=active&&active.content_type==='legacy';
  var item=readForm(wantPublish);
  if(!item.title||!item.slug){inform('Title and slug are required.',true);return;}
  if(!isLegacy&&!active&&!/^[a-z0-9][a-z0-9-]{1,98}[a-z0-9]$/.test(item.slug)){inform('Use a lowercase, hyphenated slug of at least 3 characters.',true);return;}
  if(wantPublish&&!isLegacy&&item.body_md.length<160){inform('Write at least 160 characters before publishing.',true);return;}
  if(isLegacy&&item.content_override&&item.body_md.length<160){inform('The replacement article must have at least 160 characters.',true);return;}
  busy=true;draftButton.disabled=true;publishButton.disabled=true;inform('Saving article…');
  var path=active?'/admin/api/articles/'+encodeURIComponent(active.slug):'/admin/api/articles';
  api(path,{method:active?'PUT':'POST',headers:{'Content-Type':'application/json','Accept':'application/json'},body:JSON.stringify(item)})
   .then(function(r){return refresh(r.slug).then(function(){inform(r.saved?(wantPublish?'Published version updated.':'Draft saved; published version unchanged.'):'Saved');});})
   .catch(function(e){inform('Could not save: '+e.message,true);})
   .finally(function(){busy=false;draftButton.disabled=false;publishButton.disabled=false;});
 }
 function updateMarkdown(selection){
  var editor=field('body_md'),a=editor.selectionStart,b=editor.selectionEnd,selected=editor.value.slice(a,b),text='',cursorA=a,cursorB=b;
  if(selection.wrap){
   var inner=selected||'text';
   text=selection.wrap+inner+selection.wrap;
   cursorA=a+selection.wrap.length;cursorB=cursorA+inner.length;
  }else if(selection.prefix){
   var start=editor.value.lastIndexOf('\n',a-1)+1;
   a=start;text=selection.prefix+editor.value.slice(start,b);
   cursorA=a+selection.prefix.length;cursorB=a+text.length;
  }else if(selection.code){
   text='\n'+String.fromCharCode(96,96,96)+'\n'+(selected||'// example')+'\n'+String.fromCharCode(96,96,96)+'\n';
   cursorA=a+5;cursorB=cursorA+(selected||'// example').length;
  }else if(selection.link||selection.image){
   var url=prompt('Paste an HTTPS '+(selection.image?'image URL':'link URL')+':','https://');
   if(!url||!/^https:\/\//i.test(url.trim()))return;
   var label=selected||(selection.image?'Image description':'Link text');
   text=(selection.image?'!':'')+'['+label+']('+url.trim()+')';
   cursorA=a;cursorB=a+text.length;
  }else return;
  editor.setRangeText(text,a,b,'end');
  editor.focus();
  editor.setSelectionRange(cursorA,cursorB);
  schedulePreview();
 }
 function loadHistory(){
  revisionList.replaceChildren();
  if(!active)return;
  var slug=active.slug;
  api('/admin/api/articles/'+encodeURIComponent(slug)+'/revisions').then(function(body){
   if(!active||active.slug!==slug)return;
   if(!body.revisions.length){revisionList.textContent='No previous versions saved yet.';return;}
   body.revisions.forEach(function(rev){
    var el=document.createElement('div');el.className='revision-row';
    var label=document.createElement('span');label.textContent='Saved version · '+rev.created_at;
    var button=document.createElement('button');button.type='button';button.className='btn btn-quiet';button.textContent='Load revision';
    button.addEventListener('click',function(){
     if(!confirm('Load this older version into the editor? Current unsaved changes will be replaced.'))return;
     api('/admin/api/articles/'+encodeURIComponent(slug)+'/revisions/'+rev.id).then(function(r){
      if(!active||active.slug!==slug)return;
      fieldNames.filter(function(n){return n!=='slug';}).forEach(function(n){field(n).value=safeText(r.article[n]);});
      field('content_override').checked=!!r.article.content_override;
      schedulePreview();inform('Older revision loaded. Click Save to apply it.');window.scrollTo({top:0,behavior:'smooth'});
     }).catch(function(e){inform('Revision error: '+e.message,true);});
    });
    el.appendChild(label);el.appendChild(button);revisionList.appendChild(el);
   });
  }).catch(function(e){revisionList.textContent='Version history unavailable.';});
 }
 form.addEventListener('submit',onSave);
 document.getElementById('new-article').addEventListener('click',function(){blank();field('title').focus();});
 document.getElementById('reset-form').addEventListener('click',function(){if(confirm('Clear the editor? Unsaved changes will be lost.'))blank();});
 document.getElementById('show-preview').addEventListener('click',function(){schedulePreview();document.getElementById('preview-pane').scrollIntoView({block:'nearest',behavior:'smooth'});});
 form.querySelectorAll('.toolbar button').forEach(function(b){b.addEventListener('click',function(){updateMarkdown({wrap:b.dataset.wrap,prefix:b.dataset.prefix,code:b.hasAttribute('data-wrap-code'),link:b.hasAttribute('data-insert-link'),image:b.hasAttribute('data-insert-image')});});});
 field('title').addEventListener('input',function(){if(!manualSlug&&!active){field('slug').value=slugify(field('title').value);setUrl();}});
 field('slug').addEventListener('input',function(){manualSlug=true;setUrl();});
 field('body_md').addEventListener('input',schedulePreview);
 filter.addEventListener('input',drawList);
 refresh().then(function(){blank();}).catch(function(e){inform('Cannot load article library: '+e.message,true);count.textContent='Unavailable';});
})();
