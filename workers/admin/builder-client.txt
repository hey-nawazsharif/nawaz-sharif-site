/* Nawaz CMS Visual Builder — no third-party JS; server renders escaped Markdown. */
(function(){
'use strict';
var pages=[['/index.html','Homepage'],['/about.html','About'],['/faq.html','FAQ'],['/contact.html','Contact'],['/articles.html','Articles']];
var page='/index.html',sections=[],draftVersion=0,publishedVersion=0,dirty=false,busy=false,previewTimer=null,previewReq=0,imageTarget=null,media=[];
var qs=function(id){return document.getElementById(id)};
function message(v,error){qs('message').textContent=v;qs('message').style.color=error?'#f6b3a9':'#b8eec5';}
function api(path,opts){return fetch(path,Object.assign({credentials:'same-origin',headers:{Accept:'application/json'}},opts||{})).then(function(response){
 return response.json().then(function(data){if(!response.ok)throw Error(data.error||'Request failed');return data;});
});}
function req(path,body){return api(path,{method:'POST',headers:{Accept:'application/json','Content-Type':'application/json'},body:JSON.stringify(body)});}
function el(tag,cls,text){
 var node=document.createElement(tag);if(cls)node.className=cls;if(text!==undefined)node.textContent=text;return node;
}
function randomID(){
 if(window.crypto&&typeof crypto.randomUUID==='function')return crypto.randomUUID();
 return 'section-'+Array.from(crypto.getRandomValues(new Uint8Array(10)),function(x){return (x%36).toString(36)}).join('');
}
function typeLabel(x){return {text:'Text / Markdown',image:'Image + text',cta:'Call to action',quote:'Quote',columns:'Two columns',divider:'Divider'}[x]||x;}
function selectPageButtons(){
 var host=qs('page-buttons');host.replaceChildren();
 pages.forEach(function(p){var b=el('button',p[0]===page?'current':'',p[1]);b.type='button';b.addEventListener('click',function(){
  if(dirty&&!confirm('Leave this page? Unsaved changes will be lost.'))return;
  loadPage(p[0]);
 });host.appendChild(b);});
}
function inputField(parent,section,key,label,options){
 var wrap=el('div','form-line');
 var lbl=el('label','',label);wrap.appendChild(lbl);
 var ctl;
 if(options&&options.select){
  ctl=el('select');
  options.select.forEach(function(o){var opt=el('option','',o[1]);opt.value=o[0];ctl.appendChild(opt);});
 }else if(options&&options.area){
  ctl=el('textarea');ctl.rows=options.rows||5;ctl.spellcheck=true;
 }else{ctl=el('input');ctl.type='text';}
 ctl.name=key;ctl.value=section[key]||'';ctl.maxLength=options&&options.max||6000;
 ctl.addEventListener('input',function(){section[key]=ctl.value;dirty=true;schedulePreview();});
 ctl.addEventListener('change',function(){section[key]=ctl.value;dirty=true;schedulePreview();});
 wrap.appendChild(ctl);
 if(options&&options.hint)wrap.appendChild(el('p','form-hint',options.hint));
 parent.appendChild(wrap);
 return ctl;
}
function move(from,to){
 if(to<0||to>=sections.length||from===to)return;
 var item=sections.splice(from,1)[0];sections.splice(to,0,item);dirty=true;renderList();schedulePreview();
}
function card(section,index){
 var box=el('article','section-card');box.draggable=true;
 var bar=el('div','card-head');var handle=el('span','drag-handle','⠿');handle.title='Drag to reorder';bar.appendChild(handle);
 bar.appendChild(el('strong','',String(index+1).padStart(2,'0')+' · '+typeLabel(section.type)));
 [['↑','Move up',index-1],['↓','Move down',index+1]].forEach(function(action){
  var b=el('button','mini',action[0]);b.type='button';b.title=action[1];b.disabled=action[2]<0||action[2]>=sections.length;
  b.addEventListener('click',function(){move(index,action[2]);});bar.appendChild(b);
 });
 var remove=el('button','mini remove','×');remove.type='button';remove.title='Delete section';
 remove.addEventListener('click',function(){if(!confirm('Remove this section from the draft? Published version stays unchanged until Publish.'))return;
  sections.splice(index,1);dirty=true;renderList();schedulePreview();});
 bar.appendChild(remove);box.appendChild(bar);
 box.addEventListener('dragstart',function(e){if(e.target.closest('input,textarea,select,button')){e.preventDefault();return;}e.dataTransfer.effectAllowed='move';e.dataTransfer.setData('text/plain',String(index));box.classList.add('dragging');});
 box.addEventListener('dragend',function(){box.classList.remove('dragging');});
 box.addEventListener('dragover',function(e){e.preventDefault();box.classList.add('drop-target');});
 box.addEventListener('dragleave',function(){box.classList.remove('drop-target');});
 box.addEventListener('drop',function(e){e.preventDefault();box.classList.remove('drop-target');var from=Number(e.dataTransfer.getData('text/plain'));if(Number.isInteger(from))move(from,index);});
 var fields=el('div','card-fields');
 if(section.type!=='divider'){
  inputField(fields,section,'title','Section heading',{max:160});
  if(section.type!=='image'&&section.type!=='divider'){
   inputField(fields,section,'body','Content (Markdown)',{area:true,max:6000,hint:'Use **bold**, ## headings, links and lists. All HTML is escaped.'});
  }else{
   inputField(fields,section,'body','Description (Markdown)',{area:true,max:6000,rows:3});
  }
 }
 if(['image','columns'].includes(section.type)){
  var image=inputField(fields,section,'image_url','Image URL (HTTPS)',{max:1500,hint:'Use the Media library or paste an HTTPS image URL.'});
  var use=el('button','btn tiny outline','Choose from media library');use.type='button';use.addEventListener('click',function(){imageTarget=image;qs('media-panel').open=true;qs('media-panel').scrollIntoView({block:'nearest',behavior:'smooth'});});fields.appendChild(use);
  inputField(fields,section,'image_alt','Image alternative text',{max:180});
 }
 if(['text','image','cta','columns'].includes(section.type)){
  var row=el('div','two-col');inputField(row,section,'button_label','Button label',{max:70});inputField(row,section,'button_url','Button URL / path',{max:1200});fields.appendChild(row);
 }
 if(section.type!=='divider')inputField(fields,section,'position','Placement',{select:[['bottom','After original page content'],['top','Before original page content']]});
 box.appendChild(fields);return box;
}
function renderList(){
 var list=qs('section-list');list.replaceChildren();qs('section-count').textContent=String(sections.length);
 if(!sections.length){var placeholder=el('div','empty');placeholder.appendChild(el('strong','','No custom sections yet'));placeholder.appendChild(el('p','','Use + Add to create your first WordPress-style page section.'));list.appendChild(placeholder);return;}
 sections.forEach(function(s,i){list.appendChild(card(s,i));});
}
function schedulePreview(){
 if(previewTimer)clearTimeout(previewTimer);
 var version=++previewReq;
 previewTimer=setTimeout(function(){
  var design={theme:qs('theme').value,spacing:qs('spacing').value};
  req('/admin/api/builder/preview',{sections:sections,design:design})
  .then(function(d){if(version!==previewReq)return;var style=qs('builder-preview-css');if(!style){style=el('style');style.id='builder-preview-css';document.head.appendChild(style);}style.textContent=d.css||'';
   var target=qs('render-preview');target.replaceChildren();
   if(d.html){target.innerHTML=d.html;}else target.appendChild(el('p','muted','Add sections to preview them here.'));
  }).catch(function(e){if(version!==previewReq)return;qs('render-preview').textContent='Preview unavailable: '+e.message;});
 },450);
}
function history(){
 var host=qs('history');host.textContent='Loading…';
 api('/admin/api/builder/history?page='+encodeURIComponent(page)).then(function(d){
  host.replaceChildren();if(!(d.revisions||[]).length){host.appendChild(el('p','muted','No earlier saved versions.'));return;}
  var pageAtRequest=page;
  d.revisions.forEach(function(r){
   var div=el('div','rev');div.appendChild(el('span','',r.created_at));
   var button=el('button','','Load version');button.type='button';button.addEventListener('click',function(){
    if(!confirm('Load older sections into the editor? Unsaved edits will be replaced.'))return;
    api('/admin/api/builder/history/'+r.id+'?page='+encodeURIComponent(pageAtRequest)).then(function(data){
     if(page!==pageAtRequest)return;sections=data.sections||[];dirty=true;renderList();schedulePreview();message('Revision loaded. Save draft or publish to apply.');
    }).catch(function(e){message('Revision error: '+e.message,true);});
   });div.appendChild(button);host.appendChild(div);
  });
 }).catch(function(e){host.textContent='History unavailable: '+e.message;});
}
function loadPage(next){
 page=next;sections=[];draftVersion=0;publishedVersion=0;dirty=false;busy=true;
 qs('section-list').textContent='Loading page…';selectPageButtons();
 qs('page-title').textContent=pages.find(function(x){return x[0]===next;})[1];
 qs('open-page').href='https://cms-preview.nawazsharif.in/'+next.slice(1);message('');
 api('/admin/api/builder/document?page='+encodeURIComponent(next)).then(function(d){
  if(page!==next)return;sections=d.sections||[];draftVersion=d.draft_version;publishedVersion=d.published_version;renderList();schedulePreview();history();
  qs('page-note').textContent=sections.length+' sections · '+d.published_count+' published · Drag to reorder / use arrows on mobile';
 }).catch(function(e){message('Unable to open page: '+e.message,true);qs('section-list').textContent='Unable to load sections.';})
 .finally(function(){busy=false;});
}
function add(){
 if(busy)return;if(sections.length>=20){message('Maximum 20 sections on one page.',true);return;}
 var type=qs('add-type').value;
 sections.push({id:randomID(),type:type,title:'',body:'',image_url:'',image_alt:'',button_label:'',button_url:'',position:'bottom'});
 dirty=true;renderList();schedulePreview();
 var cards=document.querySelectorAll('.section-card');if(cards.length)cards[cards.length-1].scrollIntoView({behavior:'smooth',block:'nearest'});
 message('Section added. Write content, then Save draft or Publish page.');
}
function save(intent){
 if(busy)return;
 if(intent==='publish'&&!confirm('Publish all current sections for this page on cms-preview.nawazsharif.in? The main website remains unchanged.'))return;
 busy=true;qs('save-draft').disabled=true;qs('publish').disabled=true;message('Saving '+intent+'…');
 req('/admin/api/builder/document',{page_path:page,intent:intent,sections:sections,expected_version:draftVersion})
 .then(function(data){draftVersion=data.draft_version;dirty=false;message(intent==='publish'?'Published to CMS preview. Main website remains unchanged.':'Draft saved privately. Live preview content unchanged.');history();})
 .catch(function(e){message('Save failed: '+e.message,true);})
 .finally(function(){busy=false;qs('save-draft').disabled=false;qs('publish').disabled=false;});
}
function renderMedia(){
 var target=qs('media-list');target.replaceChildren();
 media.forEach(function(item){
  var b=el('button','media-item');b.type='button';var img=el('img');img.src=item.url;img.alt='';img.loading='lazy';b.appendChild(img);b.appendChild(el('span','',item.label));
  b.addEventListener('click',function(){
   if(imageTarget){imageTarget.value=item.url;imageTarget.dispatchEvent(new Event('input',{bubbles:true}));message('Image inserted into section.');}
   else if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(item.url).then(function(){message('Image URL copied.');}).catch(function(){message('Image URL: '+item.url);});}
   else message('Image URL: '+item.url);
  });target.appendChild(b);
 });
}
function loadMedia(){api('/admin/api/builder/media').then(function(r){media=r.media||[];renderMedia();}).catch(function(){qs('media-list').textContent='Media library unavailable';});}
function saveMedia(){
 var url=qs('media-url').value.trim(),label=qs('media-label').value.trim();
 if(!/^https:\/\//i.test(url)||!label){message('Enter an HTTPS image URL and name.',true);return;}
 req('/admin/api/builder/media',{url:url,label:label}).then(function(){qs('media-url').value='';qs('media-label').value='';loadMedia();message('Image URL added to media library.');}).catch(function(e){message('Media error: '+e.message,true);});
}
function loadAppearance(){api('/admin/api/builder/appearance').then(function(r){qs('theme').value=r.draft.theme;qs('spacing').value=r.draft.spacing;schedulePreview();}).catch(function(e){message('Theme unavailable: '+e.message,true);});}
function saveAppearance(intent){
 if(intent==='publish'&&!confirm('Apply this theme to added sections on CMS preview?'))return;
 req('/admin/api/builder/appearance',{intent:intent,design:{theme:qs('theme').value,spacing:qs('spacing').value}})
 .then(function(){message(intent==='publish'?'Theme published to CMS preview.':'Theme draft saved.');schedulePreview();})
 .catch(function(e){message('Appearance error: '+e.message,true);});
}
qs('add-section').addEventListener('click',add);
qs('save-draft').addEventListener('click',function(){save('draft');});
qs('publish').addEventListener('click',function(){save('publish');});
qs('reset').addEventListener('click',function(){if(!dirty||confirm('Discard unsaved changes and reload this page?'))loadPage(page);});
qs('save-media').addEventListener('click',saveMedia);
qs('save-design').addEventListener('click',function(){saveAppearance('draft');});
qs('publish-design').addEventListener('click',function(){saveAppearance('publish');});
['theme','spacing'].forEach(function(id){qs(id).addEventListener('change',schedulePreview);});
window.addEventListener('beforeunload',function(e){if(dirty){e.preventDefault();e.returnValue='';}});
loadMedia();loadAppearance();loadPage(page);
})();
