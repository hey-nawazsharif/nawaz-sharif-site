(function(){
'use strict';
var pages=[['/index.html','Homepage'],['/about.html','About'],['/faq.html','FAQ'],['/contact.html','Contact'],['/articles.html','Articles'],['/privacy-policy.html','Privacy'],['/terms.html','Terms']];
var current='/index.html',records=[],settings=[],loading=false;
var list=document.getElementById('pages'),fields=document.getElementById('fields'),form=document.getElementById('form'),alert=document.getElementById('alert');
var title=document.getElementById('page-title'),desc=document.getElementById('page-desc'),kicker=document.getElementById('page-kicker'),preview=document.getElementById('preview'),settingsButton=document.getElementById('settings-tab');
function call(path,options){return fetch(path,Object.assign({credentials:'same-origin',headers:{Accept:'application/json'}},options||{})).then(function(r){return r.json().then(function(d){if(!r.ok)throw Error(d.error||'Request failed');return d;});});}
function notice(s,bad){alert.textContent=s;alert.style.color=bad?'#f5a9a9':'#b6e8bb';}
function createField(label,value,key,long,original,kind){
 var wrap=document.createElement('div');wrap.className='field';
 var lbl=document.createElement('label');lbl.textContent=label;var hint=document.createElement('small');hint.textContent=key==='seo_title'||key==='seo_description'?'Search & sharing':'Content';lbl.appendChild(hint);wrap.appendChild(lbl);
 var control=document.createElement(long?'textarea':'input');if(long)control.rows=key.includes('_a')?4:3;else control.type='text';control.value=value===undefined||value===null?'':String(value);control.dataset.key=key;control.maxLength=long?8000:1200;wrap.appendChild(control);
 if(original!==undefined&&value!==original){var orig=document.createElement('p');orig.className='orig';orig.textContent='Original: '+String(original).slice(0,220);wrap.appendChild(orig);}
 fields.appendChild(wrap);return control;
}
function showList(){
 list.replaceChildren();pages.forEach(function(p){
 var b=document.createElement('button');b.type='button';b.className='page-option'+(current===p[0]?' active':'');b.textContent=p[1];b.addEventListener('click',function(){select(p[0]);});list.appendChild(b);
 });settingsButton.className='page-option'+(current==='settings'?' active':'');
}
function loadHistory(){var target=document.getElementById('history');target.textContent='Loading…';call('/admin/api/site/revisions?page='+encodeURIComponent(current)).then(function(d){
 target.replaceChildren();if(!d.revisions||!d.revisions.length){target.textContent='No earlier revisions yet.';return;}
 d.revisions.forEach(function(r){var el=document.createElement('div');el.className='history-row';var text=document.createElement('span');text.textContent='Version · '+r.created_at;var b=document.createElement('button');b.type='button';b.textContent='Load draft';b.addEventListener('click',function(){if(!confirm('Restore this older revision into the editor? Unsaved changes will be lost.'))return;
 call('/admin/api/site/revisions/'+r.id+'?page='+encodeURIComponent(current)).then(function(data){
  var values=data.values||{};
  fields.querySelectorAll('[data-key]').forEach(function(input){if(Object.prototype.hasOwnProperty.call(values,input.dataset.key))input.value=String(values[input.dataset.key]);});
  notice('Previous version loaded. Click Save draft or Publish changes to apply.');
 }).catch(function(e){notice(e.message,true);});
 });el.appendChild(text);el.appendChild(b);target.appendChild(el);});
 }).catch(function(){target.textContent='Version history unavailable.';});}
function display(){
 fields.replaceChildren();var meta=pages.find(function(p){return p[0]===current;});
 kicker.textContent=current==='settings'?'GLOBAL SETTINGS':'PAGE EDITOR';
 title.textContent=meta?meta[1]+' page':'Navigation & footer';
 desc.textContent=current==='settings'?'Edit menu links using one item per line: Label | /page.html':'Only published values appear on the public CMS preview. Save draft is private.';
 preview.href='https://cms-preview.nawazsharif.in/'+(current==='settings'?'':current.slice(1));
 settingsButton.className='page-option'+(current==='settings'?' active':'');
 if(current==='settings'){
  settings.forEach(function(s){
   var value;
   try{value=JSON.parse(s.draft_json);}catch(e){value='';}
   if(s.key.endsWith('_nav'))value=Array.isArray(value)?value.map(function(x){return x.label+' | '+x.href;}).join('\n'):'';
   createField(s.label,value,s.key,true,undefined);
  });return;
 }
 records.forEach(function(r){var v=r.draft_text,lg=v.length>135||r.kind==='answer'||r.block_key.includes('intro')||r.block_key.includes('body')||r.block_key==='seo_description';
  createField(r.label,v,r.block_key,lg,r.original_text,r.kind);
 });
}
function select(page){
 if(loading)return;current=page;notice('Loading…');showList();records=[];settings=[];fields.replaceChildren();loading=true;
 var path=page==='settings'?'/admin/api/site/settings':'/admin/api/site/pages?page='+encodeURIComponent(page);
 call(path).then(function(d){if(page==='settings')settings=d.settings||[];else records=d.fields||[];display();notice('');loadHistory();})
 .catch(function(e){notice('Could not load editor: '+e.message,true);})
 .finally(function(){loading=false;});
}
function parseNav(text){
 return text.split(/\r?\n/).map(function(line){return line.trim();}).filter(Boolean).map(function(line){
  var at=line.indexOf('|');if(at<1)throw Error('Each menu line needs: Label | /page.html');
  var label=line.slice(0,at).trim(),href=line.slice(at+1).trim();
  if(!label||label.length>48||!/^\/(?!\/)[a-zA-Z0-9_./#?=&%-]*$/.test(href)||href.length>190)throw Error('Invalid menu line: '+line.slice(0,70));
  return {label:label,href:href};
 });}
form.addEventListener('submit',function(e){
 e.preventDefault();if(loading)return;
 var intent=e.submitter&&e.submitter.id==='publish'?'publish':'draft';
 if(intent==='publish'&&!confirm('Publish these changes on the public CMS preview? Production nawazsharif.in stays unchanged.'))return;
 var values={};try{fields.querySelectorAll('[data-key]').forEach(function(input){var key=input.dataset.key,val=input.value;if(current==='settings'){values[key]=key.endsWith('_nav')?parseNav(val):val.trim();}else values[key]=val.trim();});}
 catch(err){notice(err.message,true);return;}
 loading=true;for(var b of form.querySelectorAll('button'))b.disabled=true;notice('Saving '+intent+'…');
 call(current==='settings'?'/admin/api/site/settings':'/admin/api/site/pages',{
  method:'POST',headers:{'Content-Type':'application/json',Accept:'application/json'},
  body:JSON.stringify({page_path:current,intent:intent,values:values})
 }).then(function(){notice(intent==='publish'?'Changes published to CMS preview.':'Draft saved privately.');loadHistory();})
 .catch(function(err){notice('Cannot save: '+err.message,true);})
 .finally(function(){loading=false;for(var b of form.querySelectorAll('button'))b.disabled=false;});
});
settingsButton.addEventListener('click',function(){select('settings');});showList();select('/index.html');
})();
